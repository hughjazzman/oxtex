\paragraph{(c)}

Clearly, $Z_t$ is a function of $Y_0,...,Y_t$.

Since $0 < f(i)^{-N(Y_t, i)} \leq 1, E[Z_t] < \infty$.

\begin{equation}
\begin{aligned}
  E[Z_{n+1} | Y_0, ..., Y_n] 
  &= Z_n \left(\sum_{i \neq j} p(Y_n, i, j) f(i)^{-1} f(j) 
  + \Pr (\forall k\in \tau, N(Y_{n+1}, k) = N(Y_n, k)) \right)\\
  &= Z_n \left(\sum_{i\neq j}\frac{f(i)N(Y_n, i)N(Y_n, j)f(i)^{-1}f(j)}{nF(Y_n)} 
  + 1 - \sum_{i\neq j} p(Y_n, i, j)\right) \\
  &= Z_n \left(\sum_{i\neq j}\frac{f(j)N(Y_n, i)N(Y_n, j)}{nF(Y_n)} + 1 - \sum_{i\neq j} p(Y_n, i, j)\right) \\
  &= Z_n \left(\sum_{i\neq j} p(Y_n, i, j) + 1 - \sum_{i\neq j} p(Y_n, i, j)\right)\\
  &= Z_n
\end{aligned}
\end{equation}

\paragraph{(d)}
Use a "primed" version of everything 
\[f'(0) := f(0)\]
\[\forall i \in [n], f'(i) := f(1)\]
Given $Y_0, u\in [n]$, let
\[Y_{0,u}' = \begin{cases}
  0 & \text{if } Y_{0, u} = 0\\
  u & \text{otherwise}\\
\end{cases}
\]
\[\tau' := \{i \in \{0, ..., n\} | \exists u\in [n] s.t. Y_{0, u}' = i \}\]
\[F'(t) := \sum_{u\in[n]} f'(Y_{t,u}')\]
$\pi_i'$ is the probability that i wins.
By symmetry, 
\[\forall i,j\in \tau', i, j > 0, \pi_i' = \pi_j' = \pi'\]
For $i\in \{1,2\}$,
\[\pi_i = \sum_{u\in \tau': Y_{0, u}=i} \pi_u ' = N(Y_0, i) \pi'\]
\[\frac{\pi_1}{\pi_2} = \frac{N(Y_0, 1)}{N(Y_0, 2)}\]
