\section*{Question 1}

% To answer each part of the question, please simply use a paragraph command 
% \paragraph{(a)}
Suppose we have $n$ variables and $t$ clauses. 
The variables within each clause are assumed to be distinct.

Let $m_i$ be the number of variables required to be true, 
and $l_i$ be the total number of variables in clause $C_i$.
Clauses with $m_i > k$ variables that need to be true 
will not have satisfying assignments with exactly $k$ true variables.
Thus, we can ignore them and only look at clauses that need $m_i \leq k$ variables to be true.

By Theorem 4.3.5, we can use the union-of-sets sampling algorithm to create an FPRAS solution.
The set $U$ is all $2^n$ truth assignments. 
The size of the input is $t + \log |U| = t + n$.

For $1 \leq i \leq t$, the set $H_i$ is the set of truth assignments 
with exactly $k$ true variables that satisfy clause $C_i$.
Then,
\[|H_i| = \binom{n-l_i}{k-m_i}\]
as this corresponds to setting $m_i$ variables true, 
and choosing which of the remaining $(n-l_i)$ variables not in clause $C_i$ to set to be true,
which can be computed in $poly(n)$ time.

To sample uniformly from $H_i$, we can pick $(k-m_i)$ variables without replacement u.a.r.
from the $(n-l_i)$ variables not in clause $C_i$.

Finally, given a truth assignment $x\in U$, we can check in poly time whether $x \in H_i$.

\section*{Question 2}

By Theorem 4.3.5, we can use the union-of-sets sampling algorithm to create an FPRAS solution.
The set $U$ is all $n!$ permutations.
The size of the input is $t+\log |U| = t+n\log n$.

For $1 \leq i \leq t$, the set $H_i$ is the set of permutations
that are increasing on set $S_i$.
Then,
\[|H_i| = \binom{n}{k_i}(n-k_i)!=\frac{n!}{k_i!}\]
since it is the same as choosing $k_i$ elements from the set $\{1,...,n\}$, 
where only 1 of the $k_i!$ permutations of $k_i$ elements will be increasing,
multipled by the rest of the $(n-k_i)!$ permutations.
This can be computed in $poly(n)$ time.

To sample uniformly from $H_i$, we can pick $(n-k_i)$ elements without replacement from $\{1,...,n\}$ u.a.r.
The remaining $k_i$ elements will be in sorted increasing order following the indices decided by $S_i$.

Finally, given a permutation $x\in U$, we can check in poly time whether $x\in H_i$.

\paragraph{Algorithm 1}
 
\[A'(\varepsilon', \delta') = \sum_{i=1}^m A_i (\varepsilon',1-(1-\delta')^{1/m})\]

Note that 
\begin{equation}
\begin{aligned}
\left(\forall i, 1\leq i \leq m, (1-\varepsilon')x_i 
\leq A_i(\varepsilon', 1-(1-\delta')^{1/m}) \leq (1+\varepsilon') x_i\right)\\
\implies
\left((1-\varepsilon')\sum_{i=1}^m x_i \leq x' \leq (1+\varepsilon)\sum_{i=1}^m x_i\right)
\end{aligned}
\end{equation}


\begin{equation}
\begin{aligned}
    \Pr \left[x(1-\varepsilon') \leq x' \leq x(1+\varepsilon')\right]
    &= \Pr \left[(1-\varepsilon')\sum_{i=1}^m x_i \leq x' \leq (1+\varepsilon)\sum_{i=1}^m x_i\right] \\
    &\geq \prod_{i=1}^m \Pr \left[(1-\varepsilon')x_i 
    \leq A_i(\varepsilon', 1-(1-\delta')^{1/m}) \leq (1+\varepsilon') x_i\right] \\
    &= \prod_{i=1}^m \Pr \left[1-\varepsilon' 
    \leq \frac{A_i(\varepsilon', 1-(1-\delta')^{1/m})}{x_i} \leq 1+\varepsilon'\right]\\
    &\quad\quad\text{(by independence)}\\
    &= \left((1-\delta')^{1/m}\right)^m\\
    &= 1-\delta'
\end{aligned}
\end{equation}

\paragraph{Algorithm 2}

Note that 
\begin{equation}
\begin{aligned}
\left(|x'-x|\geq \varepsilon'x'\right) \implies
\left(\exists i : \left|A_i\left(\frac{\varepsilon'}{m},\frac{\delta'}{m}\right)
-x_i\right|\geq\frac{\varepsilon'x'}{m}\right)
\end{aligned}
\end{equation}

\begin{equation}
\begin{aligned}
    \Pr \left[x(1-\varepsilon') \leq x' \leq x(1+\varepsilon')\right]

    \left|A_i\left(\frac{\varepsilon'}{m},\frac{\delta'}{m}\right) - x_i \right| \geq \frac{\varepsilon'x_i}{m}\right] \\

    \quad\text{(by union bound)}\\
    &\geq 1 - m\left(\frac{\delta'}{m}\right)\\
    &= 1-\delta'
\end{aligned}
\end{equation}

This results in
\[A'(\varepsilon',\delta') = \prod_{i=1}^m A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right)\]

From part (b), we have
\[\Pr \left[\left|A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right) - x_i\right| 
\geq \frac{\varepsilon' x_i}{2m} \right]
\leq \frac{\delta'}{m}\]

Then the probability that all estimates have 
$\left|A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right) - x_i\right| 
\leq \frac{\varepsilon' x_i}{2m}$ 
is
\begin{equation}
\begin{aligned}
    \Pr \left[\forall i : \left|A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right) - x_i\right| 
    \leq \frac{\varepsilon' x_i}{2m} \right]
    &= 1 - \Pr \left[\exists i : \left|A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right) - x_i\right| 
    \geq \frac{\varepsilon' x_i}{2m} \right]\\
    &\geq 1 - \sum_{i=1}^m \Pr \left[\left|A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right) - x_i\right| 
    \geq \frac{\varepsilon' x_i}{2m} \right]
    \quad\text{(by union bound)}\\
    &\geq 1 - m\left(\frac{\delta'}{m}\right)\\
    &= 1 - \delta'
\end{aligned}
\end{equation}

From the proof of Claim 1 in Theorem 4.4.5, we then have, using 
\[\forall\varepsilon \in (0,1),1+\varepsilon\leq e^\varepsilon, 
1-\varepsilon/2\geq e^{-\varepsilon}, e^{\varepsilon/2}\leq 1 + \varepsilon,\]

\[1-\varepsilon' \leq e^{-\varepsilon'} \leq \left(1 - \frac{\varepsilon'}{2m}\right)^m 
\leq \prod_{i=1}^m \frac{A_i\left(\frac{\varepsilon'}{2m}, \frac{\delta'}{m}\right)}{x_i} 
= \frac{A'(\varepsilon',\delta')}{x}
\leq \left(1 - \frac{\varepsilon'}{2m}\right)^m 
\leq e^{\frac{\varepsilon'}{2}} \leq 1 + \varepsilon' .\]

Thus,
\[\Pr \left[1-\varepsilon' \leq \frac{A'(\varepsilon',\delta')}{x} \leq 1+\varepsilon'\right] \geq 1 - \delta'\]

From part (b) and Theorem 4.3.1, each $B_i$ is called at least 
\[k\geq \frac{3\log\frac{2}{\delta'/m}}{(\varepsilon'/2m)^2 x_i} 
= O\left(\frac{m^2 \log\frac{m}{\delta'}}{\varepsilon'^2}\right)\]
times.

\section*{Question 4}

Theorem 4.4.5 in the lecture notes is referred from when referencing claims and proofs mentioned.

Let $E=\{e_1,e_2,...,e_m\}$ denote the set of edges, and define $G_i = (V,E_i)\subseteq G$,
where $E_i=\{e_1,e_2,...,e_i\}\subseteq E$. 
Let $\Omega: G \rightarrow \{\sigma\}^*$, 
where $\Omega(G)$ denotes the full set of proper $q$-colourings for graph $G$. Then

\[|\Omega(G)|=|\Omega(G_m)|
=\frac{|\Omega(G_m)|}{|\Omega(G_{m-1})|}\cdot\dots\cdot\frac{|\Omega(G_1)|}{|\Omega(G_0)|}\cdot |\Omega(G_0)|\]

Since $G_0$ has no edges, $|\Omega(G_0)| = q^n$. Considering the ratios
\[\rho_i = \frac{|\Omega(G_i)|}{|\Omega(G_{i-1})|},\]
then
\[|\Omega(G)|=q^n\prod_{i=1}^m \rho_i\]

From the lecture notes under Claim 1, 
we just need a $(\varepsilon/(2m), \delta'/m)$-approximation for each $\rho_i$
to obtain a $(\varepsilon,\delta')$-approximation for $|\Omega(G)|$.

We then need to prove Claim 2 for $q$-colourings.
To obtain an FPRAS, we show $\exists\delta'\in (1/2,1)$ such that $\delta' \leq \rho_i \leq 1$.

The upper bound is easy since $\forall i, \Omega(G_i)\subseteq \Omega(G_{i-1})$.
For the lower bound, we show that $\rho_i \geq 3/4$.
Suppose that $e_i=(u,v)$ and denote the degree of $u,v$ in $G_i$ as $d_i(u),d_i(v)$ respectively.
Let $d_i = \min(d_i(u),d_i(v))$.
When $e_i$ is added, only $q$-colourings where $\sigma_{i-1}(u)=\sigma_{i-1}(v)$ will need to be removed.
Note that restricting the colouring of either $u$ or $v$ 
in order to have $\sigma_i(u)\neq\sigma_i(v)$, reducing the number of colours for the chosen vertex by 1,
will imply the $q$-colouring will be proper for $G_i$.
If the chosen vertex had $d+1$ choice of colours, 
then we can multiply $|\Omega(G_{i-1})|$ by a factor of $\frac{q-d}{q-d+1}$.
To ensure the restriction, we multiply $|\Omega(G_{i-1})|$ by a factor of $\frac{q-d_i}{q-d_i+1}$,
where $q-d_i$ and $q-d_i+1$ are minimum number of possible colours before and after adding edge $e_i$
(we have for $ 0 < a< b < q, \frac{q-b}{q-b+1}<\frac{q-a}{q-a+1})$.
Then we have
\[\rho_i \geq \frac{q-d_i}{q-d_i+1}\geq \frac{q-\Delta}{q-\Delta+1} \geq \frac{3}{4}\]
Thus, we have $\delta'=3/4$, similar to the proof in the lecture notes.



$\rho_i$ can then be estimated by drawing the appropriate number of samples from $\Omega(G_{i-1})$ using the PAUS,
and calculating the fraction of them that are in $\Omega(G_i)$.

Let $\mu_i$ denote the probability that the PAUS returns a sample in $\Omega(G_i)$.
Let $\hat{\rho}_i$ be the computed estimates using the PAUS.

Using the same reasoning as the lecture notes proof, fixing $\delta = \delta'/m$,
\[|\rho_i-\mu_i| \leq \frac{\varepsilon}{6m}\rho_i\]
\[|\hat{\rho}_i - \mu_i| \leq \frac{\varepsilon}{4m}\rho_i\]
\[\Pr\left[|\hat{\rho}_i - \rho_i| \leq \frac{\varepsilon}{2m}\rho_i\right] \geq 1 - \delta\]
The number of samples required is 
\[\left\lceil \frac{3\log\frac{2}{\delta}}{(\frac{\varepsilon}{8m})^2 \mu_i}\right\rceil
=O\left(\frac{m^2}{\varepsilon^2} \log\frac{1}{\delta}\right)\]

\section*{Question 5}
The Markov chain $M_t$ is \textbf{finite}, since there are $n!$ permutations.
It is \textbf{irreducible}, since you can get from any permutation to any other permutation,
in a sequence of swaps, similar to selection sort.
It is \textbf{aperiodic}, since the same card can be chosen, so the state does not change.

Thus, it converges to the unique stationary distribution $\pi$. Define
\[(\pi_1,\pi_2,...,\pi_{n!})=(1/n!,1/n!,...,1/n!)\]

Consider the system $(\pi_1,\pi_2,...,\pi_{n!})P=(\pi_1,\pi_2,...,\pi_{n!})$.
Consider $\pi_i$ on the RHS.
There are exactly $\binom{n}{2}+1$ states that could move to state $i$.
These correspond to choosing any 2 cards to swap from state $i$,
as well as self-looping choosing the exact same card.
Each have probability $1/n!$ on the left.
The self-loop has probability $1/n$ of transition to $i$, 
while the rest of the states have probability $2/n^2$.
Thus, we obtain as desired
\[\pi_i=\frac{1}{n!}\left(\binom{n}{2}\frac{2}{n^2} + \frac{1}{n}\right) = \frac{1}{n!}\]

Let $(X_t,Y_t)$ be a coupling of the Markov chain $M_t$.
We choose a coupling as follows.

\begin{itemize}
    \item Choose positions $j_1,j_2$ u.a.r. from $\{1,...,n\}$.
    \item Obtain $X_{t+1}$ from $X_t$ by swapping the card at position $j_1$ with card at position $j_2$.
    Call the card at position $j_2$ card $C$.
    \item Obtain $Y_{t+1}$ from $Y_t$ by swapping the card at position $j_1$ with card $C$.
\end{itemize}

We show that $Y_t$ is a faithful copy of the original chain.
Consider any position $k_1,k_2$. Let $C'$ be the card in position $k_2$ in $Y_t$.
Let $j_2$ be the position of $C'$ in $X_t$, and let $j_1=k_1$.
The probability that $j_1,j_2$ are chosen are each $1/n$, 
so this is the probability that position $k_1,k_2$ are chosen in the transition from $Y_t$ to $Y_{t+1}$.

Note that when a card $C$ is moved to position $j_1$ in $(X_t)$ and $(Y_t)$
it thereafter occupies the same position in both decks.
The probability that a given $C\in\{1,...,n\}$ is chosen in each transition is $1/n$.

Thus, we have a coupon collector problem. 
As in the lecture notes for the card shuffling example,
the probability that the chain has not coupled in $t$ steps is at most $\exp (-\frac{t}{n}+\log n)$.
Choosing $t=n(\log(1/\varepsilon)+\log n)$ gives that this probability is at most $\varepsilon$.
By coupling lemma, the mixing time is at most
\[n(\log(1/\varepsilon)+\log n)\]

\paragraph{(b)}
The Markov chain is \textbf{finite}, since there are $n$ states.
It is \textbf{irreducible}, since it can get from any state $i$ to $j$ by going to $0$ with a tails,
and tossing head until reaching $j$.
It is \textbf{aperiodic}, since it can use the self-loop on state $n$ 
to end up returning to each state with even or odd steps.

Thus, there is a unique stationary distribution $\pi$. Define
\[\pi=(1/2,1/2^2,1/2^3,...,1/2^{n-1},1/2^n,1/2^n)\]
so $\pi_i=1/2^{i+1}$ for $0\leq i<n$.
Consider $\pi_i$.
\[\forall i, 0 < i < n, \pi_i = \frac{1}{2}\pi_{i-1} = \frac{1}{2^{i+1}}\]
\[\pi_0 = \frac{1}{2}\sum_{i=0}^n \pi_i = \frac{1}{2}\]
\[\pi_n = \frac{1}{2}(\pi_{n-1}+\pi_n) = \frac{1}{2^n}\]
Then $\pi=\pi P$. The chosen $\pi$ is the unique stationary distribution.

\paragraph{(c)}
We define a coupling $(X_t,Y_t)$ by having $(X_t)$ and $(Y_t)$ behave like independent copies 
of the Markov chain if $X_t\neq Y_t$, and otherwise have them behave identically.
Note that
\[\forall i,j,i\neq j, \Pr[(X_{t+1},Y_{t+1})=(0,0)|(X_t,Y_t)=(i,j)] = 1/4\]
Then
\[\Pr[X_{t+1}=Y_{t+1}|X_t\neq Y_t] \geq 1/4\]

The probability of not being equal after $t$ steps is at most $(3/4)^t$.
Set $\varepsilon\geq(3/4)^t$, so $t\geq\log_{4/3}(1/\varepsilon)$ and
\[\Pr[X_t\neq Y_t]\leq \varepsilon\]
By the coupling lemma, the mixing time is bounded by $O(\log(1/\varepsilon))$.

\paragraph{(d)}
Note that this Markov chain is also finite, irreducible, and aperiodic, since it is the time-reversal.
Thus, it has a unique stationary distribution.
In fact, its stationary distribution is exactly the same as before.
\[\hat{\pi}=\pi=(1/2,1/2^2,1/2^3,...,1/2^{n-1},1/2^n,1/2^n)\]

Observe that 
\begin{itemize}
    \item $\forall i<n, \hat{P}_{0,i}=\frac{1}{2^{i+1}}$
    \item $0\leq i < n-1,\hat{P}_{i+1,i}=1$
    \item $\hat{P}_{n,n-1}=\hat{P}_{n,n}=\frac{1}{2}$ 
\end{itemize}

Analysing the $\hat{\pi}_i$,
\[\forall i, 0<i<n, \hat{\pi}_i = \hat{\pi}_{i+1} + \frac{1}{2^{i+1}}\hat{\pi}_0
=\frac{2}{2^{i+2}}=\frac{1}{2^{i+1}}=\hat{\pi}_i\]
\[\hat{\pi}_0 = \frac{1}{2}(\hat{\pi}_0+\hat{\pi}_1)=\frac{1}{2}=\hat{\pi}_0\]
\[\hat{\pi}_n = \frac{1}{2}\hat{\pi}_n + \frac{1}{2^n}\hat{\pi}_0 
=\frac{2}{2^{n+1}}=\frac{1}{2^n}=\hat{\pi}_n\]
Then $\hat{\pi}=\hat{\pi}\hat{P}$.

Note that since for $0\leq i<n-1, \hat{P}_{i+1,i}=1$ then for $0<i<n, \hat{P}_{i,0}^i=1$.

Define $\vec{1}_i= (0,...,1,...,0)$ where it is equal to 1 at index $i$, and 0 otherwise.

Consider an arbitrary $i, 0\leq i<n$, such that $\hat{\pi}_B^0=\vec{1}_i$.
Then since $\hat{P}_{i,0}^i=1$, we have $\hat{\pi}_B^i=\vec{1}_0$. 
Then we must have $\hat{\pi}_B^{i+1}=\hat{\pi}$, where clearly $i+1\leq n$.
Since it is in a stationary distribution, it will stay there so $\hat{\pi}_B^{n}=\hat{\pi}$.

For the case $i=n$ where $\hat{\pi}_B^0=\vec{1}_n$, note that in each time step $t\leq n$, for $i<n$,
\[(i+t\geq n) \implies \hat{\pi}_{B,i}^t = \frac{1}{2^{i+1+t-n}}\]
and $\hat{\pi}_{B,n}^t=\frac{1}{2^t}$.

Then at time step $t=n$, we have
\[\hat{\pi}_{B,i}^n = \frac{1}{2^{i+1}}\]
\[\hat{\pi}_{B,n}^n=\frac{1}{2^n}\]

Thus, $\hat{\pi}^n=\hat{\pi}$.
We have 
\[\forall i, \hat{\pi}_B^0=\vec{1}_i \implies \hat{\pi}_B^n=\hat{\pi}\]

Next, we note that for any distribution $\hat{\pi}_D^0=(p_1,p_2,...,p_n)$,
it can be expressed in terms of $\vec{1}_i$ as
\[\hat{\pi}_D^0=(p_0,p_1,...,p_n)=\sum_{i=0}^n p_i \vec{1}_i\]
where $\sum_{i=0}^n p_i =1$.
Since after $t=n$ steps, we know that $\vec{1}_i$ will be in the stationary distribution $\hat{\pi}$,
\[\hat{\pi}_D^n = \sum_{i=0}^n p_i \hat{\pi} = \hat{\pi}\]

Then the mixing time is at most $n$ for any distribution of $\hat{\pi}_D$.
