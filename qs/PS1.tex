\begin{proposition}
    \textbf{Rel} composition satisfies the associativity axiom.
\end{proposition}
\begin{proof}
    Let $Q:W\rightarrow X, R:X\rightarrow Y, S: Y\rightarrow Z$ be arbitrary relations, 
    so $Q\subseteq W\times X, R\subseteq X \times Y, S\subseteq Y \times Z$.
    Then 
    \begin{equation}
    \begin{aligned}
        S\circ (R\circ Q)(w, z) &\iff \exists y: (R\circ Q)(w,y) \wedge S (y, z)\\
        &\iff \exists y: (\exists x: Q(w,x) \wedge R(x,y)) \wedge S(y,z)\\
        &\iff \exists x,y: Q(w,x) \wedge R(x,y) \wedge S(y,z)\\
        &\iff \exists x,y: Q(w,x) \wedge (R(x,y) \wedge S(y,z))\\
        &\iff \exists x: Q(w,x) \wedge (\exists y: R(x,y) \wedge S(y,z))\\
        &\iff \exists x: Q(w,x) \wedge (S\circ R(x, z))\\
        &\iff (S\circ R) \circ Q(w,z)
    \end{aligned}
    \end{equation}
\end{proof}
\begin{proposition}
    \textbf{Rel} composition satisfies the identity axiom.
\end{proposition}
\begin{proof}
    Let $R: X\rightarrow Y$ be an arbitrary relation. By definition of the identity relation, for set $Z$,
    \[\text{id}_Z = \{(a,a)|a\in Z\}\]
    so
    \[R\circ\text{id}_X=R=\text{id}_Y\circ R\]
\end{proof}

\section*{Question 3}
\addtocounter{section}{1}
\begin{proposition}
    In \textbf{Set}, an arrow is epic if and only if it is surjective.
\end{proposition}
\begin{proof}
    Suppose $f:X\rightarrow Y$ is surjective, and that $g\circ f=h\circ f$, where $g,h:Y\rightarrow Z$.
    By surjectivity,
    \[\forall y\in Y, \exists x\in X: f(x)=y\]
    Applying either $g$ or $h$, we have
    \[\forall y\in Y, \exists x\in X: g\circ f(x)=g(y)\]
    \[\forall y\in Y, \exists x\in X: h\circ f(x)=h(y)\]
    Since $g\circ f=h\circ f$,
    \[\forall y\in Y, \exists x\in X: g\circ f(x)=g(y)=h\circ f(x)=h(y)\]
    \[\forall y\in Y: g(y)=h(y)\]
    so $g=h$.
    We have 
    \[\forall g,h: g\circ f = h\circ f \Rightarrow g = h\]
    so $f$ is epic.

    Suppose $f$ is not surjective. Then
    \[\exists y\in Y,\forall x\in X: f(x)\neq y\]
    Let $y'\in Y$ be such an element.
    Define $Z=\{0,1\}$, where $g=0$, and $\forall y\in Y-\{y'\},h(y)=0$, with $h(y')=1$.
    Clearly, $g\neq h$ since $g(y')\neq h(y')$.
    Then we have
    \[\forall x\in X: g\circ f(x) = h\circ f(x)=0\wedge f(x)\neq y'\]
    but $g\neq h$.
    Thus, $f$ is not epic.

\end{proof}

\section*{Question 4}
\addtocounter{section}{1}
\paragraph{(a)}
\begin{proposition}
    If $f,g$ are arrows such that $g\circ f$ is monic, then $f$ is monic.
\end{proposition}
\begin{proof}
    Suppose $g\circ f$ is monic and for arbitrary arrows $h',k'$ (where the codomain matches $f$),
    \[f\circ h' = f\circ k'\]
    Since $g\circ f$ is monic, we have
    \[\forall h,k: g\circ f\circ h = g\circ f \circ k \implies h=k\]
    Composing $g$ to the first equation,
    \[g\circ f\circ h' = g\circ f\circ k'\]
    Using $g\circ f$ being monic, then $h'=k'$.
    Thus, with $h',k'$ being arbitrary, we have that
    \[\forall h,k:  f\circ h = f \circ k \implies h=k\]
    So $f$ is monic.
\end{proof}

\subsection*{(b)}
\begin{proposition}
    If $f$ is split epic, then it is epic.
\end{proposition}
\begin{proof}
    Suppose $f$ is split epic, and for arbitrary arrows $h',k'$ (where the domain matches $f$),
    \[h'\circ f = k'\circ f\]
    Since $f$ is split epic,
    \[\exists g: g\circ f=\text{id}\]
    Precomposing $g$ to the first equation,
    \[h'\circ f\circ g = k'\circ f\circ g\]
    \[\implies h'\circ \text{id} = k'\circ \text{id}\]
    \[\implies h' = k'\]
    Since $h',k'$ are arbitrary, we have that
    \[\forall h,k: k\circ f = h\circ f \implies h = k\]
    So $f$ is epic.
\end{proof}
\subsection*{(c)}
\begin{proposition}
    If $f$ and $g\circ f$ are iso, then $g$ is iso.
\end{proposition}
\begin{proof}
    Suppose $f$ and $g\circ f$ are iso. Let $f:A\rightarrow B,g:B\rightarrow C$.
    Then
    \[\exists h: h\circ f = \text{id}_A \wedge f\circ h = \text{id}_B\]
    \[\exists h': h'\circ g\circ f = \text{id}_A \wedge g\circ f\circ h' = \text{id}_C\]
    Then we can show that $h'\circ g = h$, by composing $h'\circ g\circ f = \text{id}_A$ with $h$,
    \[h'\circ g\circ f \circ h = \text{id}_A \circ h\]
    \[\implies h'\circ g\circ \text{id}_A = h\]
    \[\implies h'\circ g= h\]
    Define $h''=f\circ h'$. Then we have
    \[h'' \circ g = f\circ h' \circ g = f\circ h = \text{id}_B\]
    \[g \circ h'' = g\circ f\circ h' = \text{id}_C\]
    So $g$ is iso.
\end{proof}
\subsection*{(d)}
\begin{proposition}
    If $f$ is monic and split epic, the it is iso.
\end{proposition}
\begin{proof}
    Let $f:A\rightarrow B$ be monic and split epic. Since $f$ is split epic,
    \[\exists k: f\circ k = \text{id}_B\]
    Suppose $k\circ f\neq \text{id}_A$.
    Let $g=k\circ f,h=\text{id}_A$ so $g\neq h$. Then
    \[f\circ g = f\circ k\circ f = \text{id}_B \circ f = f\]
    \[f\circ h = f\circ \text{id}_A = f\]
    So $f\circ g = f\circ h = f$ but $g \neq h$.
    Thus $f$ is not monic, a contradiction.
    So we must have $k\circ f= \text{id}_A$.
    As $k\circ f= \text{id}_A$ and $f\circ k = \text{id}_B$, $f$ is iso.
\end{proof}

\section*{Question 5}
\addtocounter{section}{1}
\subsection*{(a)}
\begin{proposition}
    $i$ is a monomorphism and an epimorphism.
\end{proposition}
\begin{proof}
    Let $(G,\cdot,1_G)$ be an arbitrary monoid where we have arrows $g,h:(G,\cdot,1_G)\rightarrow(\mathbb{N},+,0)$,
    and $i\circ g=i\circ h$.
    Since $i$ is an inclusion map, its underlying function is the identity on $\mathbb{N},|i|=\text{id}_\mathbb{N}$.
    We have
    \[\forall k\in G:|i\circ g|(k)=|i\circ h|(k)\]
    \[\implies\forall k\in G:|i| \circ |g|(k)=|i|\circ |h|(k)\]
    \[\implies\forall k\in G: |g|(k)= |h|(k)\]
    Then $g=h$, so $i$ is monic.

    Similarly, let $(G',\cdot,1_{G'})$ be an arbitrary monoid 
    where we have arrows $g',h':(\mathbb{Z},+,0)\rightarrow(G',\cdot,1_{G'})$,
    and $g'\circ i=h'\circ i$. 
    As before,
    \[\forall n\in \mathbb{N}:|g\circ i|(n)=|h\circ i|(n)\]
    \[\implies\forall n\in \mathbb{N}:|g| \circ |i|(n)=|h|\circ |i|(n)\]
    \begin{equation}
        \implies\forall n\in \mathbb{N}: |g|(n)= |h|(n)
        \label{eq:1}
    \end{equation}
    To prove the equality for $n<0$, we use the structure preserving nature of monoid homomorphisms for $g',h'$.
    By definition,
    \[\forall n_1,n_2\in\mathbb{Z}:|g'|(n_1+n_2)=|g'|(n_1)\cdot |g'|(n_2) \wedge |g'|(0)=1_{G'}\]
    \[\forall n_1,n_2\in\mathbb{Z}:|h'|(n_1+n_2)=|h'|(n_1)\cdot |h'|(n_2) \wedge |h'|(0)=1_{G'}\]
    Then we see that $|g'|(0)=|h'|(0)=1_{G'}$, and if we set $n_2=-n_1$, we have
    \begin{equation}
        \forall n\in\mathbb{Z}:|g'|(0)=|g'|(n)\cdot |g'|(-n)=1_{G'}
        \label{eq:2}
    \end{equation}
    \[\forall n\in\mathbb{Z}:|h'|(0)=|h'|(n)\cdot |h'|(-n)=1_{G'}\]
    By symmetry, since $n\in\mathbb{Z}$, we also have 
    \[\forall n\in\mathbb{Z}:|g'|(0)=|g'|(-n)\cdot |g'|(n)=1_{G'}\]
    \begin{equation}
        \forall n\in\mathbb{Z}:|h'|(0)=|h'|(-n)\cdot |h'|(n)=1_{G'}
        \label{eq:3}
    \end{equation}
    \[\]
    If $n>0$, then $|g|(n)= |h|(n)$ from (\ref{eq:1}), so using (\ref{eq:2}),
    \[\forall n\in\mathbb{N}:|h'|(n)\cdot |g'|(-n)=1_{G'}\]
    Preoperating with $|h'|(-n)$,
    \[\forall n\in\mathbb{N}:|h'|(-n)\cdot |h'|(n)\cdot |g'|(-n)=|h'|(-n)\cdot 1_{G'}\]
    \[\implies \forall n\in\mathbb{N}:1_{G'}\cdot |g'|(-n)=|h'|(-n) \quad\text{by (\ref{eq:3})}\]
    \[\implies \forall n\in\mathbb{N}:|g'|(-n)=|h'|(-n)\]
    Thus, we have
    \[\forall n\in\mathbb{Z}:|g'|(n)=|h'|(n)\]
    so $g'=h'$, and $i$ is epic.
\end{proof}

\section*{Question 6}
\addtocounter{section}{1}
\subsection*{(a)}
\begin{proof}
    Suppose we have arrows $k_1:A'\rightarrow A,k_2:B \rightarrow B'$, which exist by definition.
    Their unique inverses $k_1^{-1}:A\rightarrow A',k_2^{-1}:B' \rightarrow B$ also exist since the arrows are iso.

    We can define a mapping between \textit{homsets} $h:\mathcal{G}(A,B)\rightarrow \mathcal{G}(A',B')$ where
    \[\forall i\in\mathcal{G}(A,B):h(i)=k_2\circ i\circ k_1\]
    This mapping is injective,
    \[\forall i,j \in \mathcal{G}(A,B): h(i)=h(j)\]
    \[\implies \forall i,j \in \mathcal{G}(A,B):  k_2\circ i \circ k_1 = k_2\circ j\circ k_1\]
    \[\implies \forall i,j \in \mathcal{G}(A,B):  
    k_2^{-1}\circ k_2\circ i \circ k_1 \circ k_1^{-1}= k_2^{-1}\circ k_2\circ j\circ k_1\circ k_1^{-1}\]
    \[\implies \forall i,j \in \mathcal{G}(A,B):  
    \text{id}_B\circ i \circ \text{id}_A = \text{id}_B\circ j\circ \text{id}_A\]
    \[\implies \forall i,j \in \mathcal{G}(A,B):  i=j\]

    For surjectivity, for each arrow $i'\in\mathcal{G}(A',B')$, define $i=k_2^{-1}\circ i' \circ k_1^{-1}$ so
    \[h(i)=k_2\circ k_2^{-1}\circ i' \circ k_1^{-1} \circ k_1 
    = \text{id}_{B'}\circ i' \circ \text{id}_A = i'\]

    Thus, $h$ is bijective. All homsets have the same cardinality.


\end{proof}
