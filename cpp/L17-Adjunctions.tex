\begin{example}

\[(B\Rightarrow g)(h) := g\circ h\]

	&&& {B\Rightarrow Y} \\
	\\
	&&&& {f(x)=(\hat{f}\circ -)(\lambda b.(x,b))}
	\arrow["{\eta_X}", from=2-3, to=2-4]
