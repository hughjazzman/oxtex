\begin{claim}
    $h$ is monotone.
\end{claim}
\begin{proof}
Suppose $x,y\in P, x\leq y$. Then
\begin{equation}
\begin{aligned}
    h(x) &\leq \bigvee \{h(x),h(y)\}\quad\text{(by LUB)}\\
    &= h\left(\bigvee \{x,y\}\right)\\
    &= h(y) \quad\text{(since $y = \bigvee \{x,y\}$)}
\end{aligned}
\end{equation}
So $h$ is monotone.
\end{proof}

\begin{proposition}
  $U$ has a left adjoint.
\end{proposition}
\begin{proof}
Define the functor $F:\textbf{Set} \rightarrow \textbf{SL}$
\[F(X) = (\mathcal{P}(X), \subseteq)\]
\[F(f: X\rightarrow Y) : \mathcal{P}(X) \rightarrow \mathcal{P}(Y) := A \mapsto \{f(a) | a\in A\}\]
which is the covariant powerset functor for arrows, so functoriality is satisfied.

Note that $\bigvee S = \bigvee \{A | A\in S\} = \bigvee_{A\in S} A$.
The functor on arrows preserves least upper bounds where $\forall f:X\rightarrow Y$ in \textbf{Set},
\begin{equation}
\begin{aligned}
  F(f)\left(\bigvee S\right) &= F(f)\left(\bigcup_{A\in S} A\right)\\
  &= \left\{f(a) \middle| a \in \bigcup_{A\in S} A\right\}\\
  &= \bigcup_{A\in S} \{f(a) | a \in A\}\\
  &= \bigcup_{A\in S} F(f)(A)\\
  &= \bigvee_{A\in S} F(f)(A)\\
  &= \bigvee \{ F(f)(A) | A\in S\}
\end{aligned}
\end{equation}

\begin{claim}
  $F:\textbf{Set} \rightarrow \textbf{SL}$ as defined is the left adjoint of $U$.
\end{claim}
$F$ is the left adjoint of $U$, iff $\forall A\in\text{Ob}(\textbf{Set})$,
there exists a unit $\eta_A : A\rightarrow UFA$ such that 
\[\forall f: A\rightarrow B,\exists! \hat{f}:\mathcal{P}(A) \rightarrow B\]
where $f\in\text{Ar}(\textbf{Set}),\hat{f}\in\text{Ar}(\textbf{SL})$, the following diagram commutes

\[\eta_C(c) = \{c\}\]
From an arrow $f:A\rightarrow B$ in \textbf{Set}, we can construct an arrow in \textbf{SL}
\[\hat{f} : \mathcal{P}(A) \rightarrow B := X \mapsto \bigvee F(f)(X)\]
where $U(\hat{f}) = \hat{f}$.
Then $\forall a\in A$,
\[(U(\hat{f}) \circ \eta_A) (a) = \hat{f} (\{a\}) = f(a)\]
So $f = U(\hat{f}) \circ \eta_A$.

Suppose there's another $g: \mathcal{P}(A) \rightarrow B$ such that $f = U(g) \circ \eta_A = g\circ\eta_A$.
Then it also preserves least upper bounds where $\forall X\subseteq A$,
\begin{equation}
\begin{aligned}
  g(X) &= g\left(\bigvee \{\{a\} | a\in X\}\right)\\
  &= \bigvee \{g(\{a\}) | a\in X\} \quad \text{(by def. of arrow in SL)}\\
  &= \bigvee \{ g\circ \eta_A (a) | a\in X\} \quad\text{(since $\eta_A(a)=\{a\}$)}\\
  &= \bigvee \{f(a) | a\in X\}\\
  &= \bigvee F(f)(X)\\
  &= \hat{f}(X)
\end{aligned}
\end{equation}
so $g=\hat{f}$ is unique.
Thus, $F$ is the left adjoint of $U$.
\end{proof}

\begin{proposition}
  The contravariant powerset functor $\mathcal{P}:\textbf{Set}^{\text{op}}\rightarrow\textbf{Set}$
  is its own left adjoint.
\end{proposition}
\begin{proof}
$\mathcal{P}^{op}$ is the left adjoint of $\mathcal{P}$ iff $\forall X\in\text{Ob}(\textbf{Set})$, there exists a unit
\[\eta_X : X \rightarrow X\]
such that $\forall f: X\rightarrow \mathcal{P}(Y)$ in \textbf{Set},
$\exists! \hat{f}: \mathcal{P}^{op}(X) \rightarrow Y$ in $\textbf{Set}^{\text{op}}$
where the diagram commutes
