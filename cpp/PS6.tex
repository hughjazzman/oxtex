\section*{Question 1}
\addtocounter{section}{1}
\begin{claim}
    $h$ is monotone.
\end{claim}
\begin{proof}
Suppose $x,y\in P, x\leq y$. Then
\begin{equation}
\begin{aligned}
    h(x) &\leq \bigvee \{h(x),h(y)\}\quad\text{(by LUB)}\\
    &= h\left(\bigvee \{x,y\}\right)\\
    &= h(y) \quad\text{(since $y = \bigvee \{x,y\}$)}
\end{aligned}
\end{equation}
So $h$ is monotone.
\end{proof}
