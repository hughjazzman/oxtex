\begin{claim}
    $h$ is monotone.
\end{claim}
\begin{proof}
Suppose $x,y\in P, x\leq y$. Then
\begin{equation}
\begin{aligned}
    h(x) &\leq \bigvee \{h(x),h(y)\}\quad\text{(by LUB)}\\
    &= h\left(\bigvee \{x,y\}\right)\\
    &= h(y) \quad\text{(since $y = \bigvee \{x,y\}$)}
\end{aligned}
\end{equation}
So $h$ is monotone.
\end{proof}

\begin{proposition}
  $U$ has a left adjoint.
\end{proposition}
\begin{proof}
Define the functor $F:\textbf{Set} \rightarrow \textbf{SL}$
\[F(X) = (\mathcal{P}(X), \subseteq)\]
\[F(f: X\rightarrow Y) : \mathcal{P}(X) \rightarrow \mathcal{P}(Y) := A \mapsto \{f(a) | a\in A\}\]
which is the covariant powerset functor for arrows, so functoriality is satisfied.

Note that $\bigvee S = \bigvee \{A | A\in S\} = \bigvee_{A\in S} A$.
The functor on arrows preserves least upper bounds where $\forall f:X\rightarrow Y$ in \textbf{Set},
\begin{equation}
\begin{aligned}
  F(f)\left(\bigvee S\right) &= F(f)\left(\bigcup_{A\in S} A\right)\\
  &= \left\{f(a) \middle| a \in \bigcup_{A\in S} A\right\}\\
  &= \bigcup_{A\in S} \{f(a) | a \in A\}\\
  &= \bigcup_{A\in S} F(f)(A)\\
  &= \bigvee_{A\in S} F(f)(A)\\
  &= \bigvee \{ F(f)(A) | A\in S\}
\end{aligned}
\end{equation}

\begin{proposition}
  If $f$ has a right adjoint, then $f$ preserves least upper bounds.
\end{proposition}
\begin{proof}
Suppose $f$ has a right adjoint $g: Q\rightarrow P$.
\begin{equation}
\begin{aligned}
  &\forall x\in S\subseteq P, x \leq \bigvee S \quad\text{(by upper bound)}\\
  &\implies \forall x\in S, f(x) \leq f\left(\bigvee S\right) \quad\text{($f$ is monotone)}\\
  &\implies \bigvee\{f(x) | x\in S\} \leq f\left(\bigvee S\right) \quad\text{(by \textit{least} UB)}
\end{aligned}
\end{equation}
\begin{equation}
\begin{aligned}
  &\forall x\in S\subseteq P, f(x) \leq \bigvee\{f(x) | x\in S\} \quad\text{(by upper bound)}\\
  &\implies \forall x\in S, x \leq g\left(\bigvee\{f(x) | x\in S\}\right) \quad\text{($g$ is right adjoint)}\\
  &\implies \bigvee S \leq g\left(\bigvee\{f(x) | x\in S\}\right) \quad\text{(by \textit{least} UB)}\\
  &\implies f\left(\bigvee S\right) \leq \bigvee\{f(x) | x\in S\} \quad\text{($f$ is left adjoint)}
\end{aligned}
\end{equation}
Then since $Q$ is a poset, these imply that
\[f\left(\bigvee S\right) = \bigvee\{f(x) | x\in S\}\]
  
\end{proof}
