\begin{itemize}
    \item \textbf{faithful} if each map $F_{A,B}:\mathcal{C}(A,B) \rightarrow \mathcal{D}(F\ A, F\ B)$ is injective;
    \item \textbf{full} if each map $F_{A,B}:\mathcal{C}(A,B) \rightarrow \mathcal{D}(F\ A, F\ B)$ is surjective
\end{itemize}

\begin{proposition}
    $\forall f\in \text{Ar}(\textbf{Set}),(S\ f):\mathcal{C}\rightarrow \mathcal{D}$ satisfies functoriality.
\end{proposition}
\begin{proof}
Since there is a valid function where $\forall a\in C,f(a)=b$,
$(S\ f)$ will map every object $(S\ a)$ in $\mathcal{C}$ to an $(S\ b)$ in $\mathcal{D}$.

There are no arrows between objects in the discrete categories,
only identity arrows. For any identity arrow $1_A$ in $\mathcal{C}$,
\[(S\ f) (1_A\circ 1_A) = (S\ f)(1_A) = 1_{(S\ f) (1_A)} = (S\ f) (1_A)\circ(S\ f) (1_A)\]
\end{proof}

\begin{fact}
    The functor $S:\textbf{Set}\rightarrow \textbf{Cat}$ satisfies functoriality.
\end{fact}

\begin{proof}
    Every object set $A$ in \textbf{Set} will get mapped to a discrete category $S\ A$ in \textbf{Cat}.
    For every pairs of functions $f:A\rightarrow B,g:B\rightarrow C$ in \textbf{Set},
    and $a\in A,c\in C$,
\begin{equation}
\begin{aligned}
    (S (g\circ f)) :: S\ a \mapsto S\ c &\iff g\circ f(a) = c\\
    &\iff \exists b\in B: f(a) = b \wedge g(b) = c\\
    &\iff \exists b\in B: S\ f:: S\ a \mapsto S\ b \wedge S\ g:: S\ b \mapsto S\ c\\
    &\iff (S\ g \circ S\ f):: S\ a \mapsto S\ c
\end{aligned}
\end{equation}
    \[S\ 1_A :: S\ a \mapsto S\ a = 1_{S\ A}\]
    Then functoriality is satisfied.
\end{proof}

\begin{proposition}
    The functor $S:\textbf{Set}\rightarrow \textbf{Cat}$ is full and faithful.
\end{proposition}

\begin{proof}
To show faithfulness, let $f,g:C\rightarrow D$ be arbitrary functions in \textbf{Set} 
so $S\ f, S\ g: \mathcal{C} \rightarrow \mathcal{D}$.
Suppose $S\ f=S\ g$. Then since for all $a\in C$,
\[S\ f::S\ a \mapsto S\ b \iff f::a\mapsto b\]
\[S\ g::S\ a \mapsto S\ b \iff g::a\mapsto b\]
we have 
\[\forall a\in C:f::a\mapsto b \iff S\ f=S\ g::S\ a \mapsto S\ b \iff g::a\mapsto b\]
and $f=g$. $S$ is faithful.

For fullness, let $F:\mathcal{C}\rightarrow \mathcal{D}$ be an arbitrary functor in \textbf{Cat}$(S\ A, S\ B)$,
where $A,B$ are sets in \textbf{Set}.
Let $f:A\rightarrow B$.
For each object $S\ a$ in $S\ A$ where $F :: S\ a \mapsto S\ b$, 
we can have some $a\in A, f(a)=b$ since $S\ A$ has $|A|$ objects.
Then $S\ f=F$.
So $S$ is full.

\end{proof}

\begin{definition}
The bivariant Hom-functor is given by
\[\mathcal{C}(\_,\_)(A,B):=\mathcal{C}(A,B)\]
\[\mathcal{C}(\_,\_)(g:B\rightarrow A,f:C\rightarrow D):= h\mapsto f\circ h\circ g,\forall h\in\mathcal{C}(A,C)\]

for all objects and arrows $A,B,C,D\in\text{Ob}(\mathcal{C}),f,g\in\text{Ar}(\mathcal{C})$.
\end{definition}

\begin{proposition}
The bivariant Hom-functor is a functor.
\end{proposition}

\begin{proof}
Let $A,B,C,D,E,F\in \text{Ob}(\mathcal{C})=\text{Ob}(\mathcal{C}^{\text{op}})$ be arbitrary objects.

For any object pair $A,B$, the functor has a mapping to the set $\mathcal{C}(A,B)$.

Let $f:A\rightarrow B, g:B\rightarrow C$ be arbitrary arrows in $\mathcal{C}$,
and $h:E\rightarrow D, k:F\rightarrow E$ be arbitrary arrows in $\mathcal{C}^{\text{op}}$.
\begin{equation}
\begin{aligned}
\mathcal{C}(\_,\_)(h,g)\circ\mathcal{C}(\_,\_)(k,f) 
&= (p \mapsto g \circ p\circ h)\circ(p'\mapsto f \circ p'\circ k)\\
&= (p \mapsto g \circ (f\circ p \circ k) \circ h)\\
&= (p \mapsto (g \circ f)\circ p \circ (k \circ h))\\
&= \mathcal{C}(\_,\_)(h\circ k,g\circ f)\\
&= \mathcal{C}(\_,\_)((h,g)\circ(k,f))
\end{aligned}
\end{equation}
So the composition law is satisfied.

For the identity laws,
\begin{equation}
\begin{aligned}
\mathcal{C}(\_,\_)(1_A,1_B)&= \mathcal{C}(1_A,1_B)\\
&= h\mapsto 1_B \circ h \circ 1_A\\
&= h \mapsto h\\
&= 1_{\mathcal{C}(A,B)}\\
&= 1_{\mathcal{C}(\_,\_)(A,B)}\\
\end{aligned}
\end{equation}
Thus, the bivariant Hom-functor satisfies \textit{functoriality}.

\end{proof}

    By fullness, for any arrow $g$ in $\mathcal{D}$, there exists an arrow $f$ in $\mathcal{C}$

    Using fullness we find $f:A\rightarrow B$,{\color{red}\sout{$f':B' \rightarrow A'$}}

    \[g^{-1}\circ g = 1_{F\ A},\quad g\circ g^{-1} = 1_{F\ B}\]
    By functoriality, we have from composition that 
    \[F(f\circ f')=F\ f\circ F\ f' = g \circ g^{-1}=1_{F\ B}\]

    Given an iso arrow $F\ f$ in $\mathcal{D}$, we then have an iso arrow $f$ in $\mathcal{C}$.
    The full and faithful functor reflects isomorphisms.

\end{proof}

\subsection*{(c)}
\begin{proposition}
    Every equivalence preserves monics.
\end{proposition}

\begin{proof}
    Let $F:\mathcal{C}\rightarrow \mathcal{D}$ be an equivalence, 
    and let $f:B\rightarrow C$ be a monic arrow in $\mathcal{C}$.
    Then $F\ f:F\ B\rightarrow F\ C$.
    Using fullness, let $F\ g, F\ h:A'\rightarrow F\ B$ be arbitrary arrows in $\mathcal{D}$.

    By equivalence, there exists an object $A$ of $\mathcal{C}$ such that $F(A)\cong A'$.

    By fullness, let the arrows $g,h:A\rightarrow B$ be in $\mathcal{C}$.

    Suppose $F\ f\circ F\ g = F\ f\circ F\ h$. Then
\begin{equation}
\begin{aligned}
    F\ f\circ F\ g = F\ f\circ F\ h &\implies F(f\circ g)=F(f\circ h) &&\quad \text{(by functoriality)}\\
    &\implies f\circ g = f\circ h &&\quad\text{(by faithfulness)}\\
    &\implies g=h &&\quad\text{(since $f$ is monic)}\\
    &\implies F\ g=F\ h
\end{aligned}
\end{equation}
    Then
    \[\forall F\ g, F\ h:F\ f\circ F\ g = F\ f\circ F\ h\implies F\ g= F\ h\]
    so $F\ f$ is monic and the equivalence preserves monics.
\end{proof}

\begin{proposition}
    If $ABCD,BEDF$ are pullback squares, then so is $AECF$.
\end{proposition}

	\\

	\\
	& C && D && F
	\arrow["a", from=3-2, to=3-4]
	\arrow["b", from=3-4, to=3-6]
	\arrow["e", from=3-6, to=5-6]
	\arrow["d", from=3-4, to=5-4]
	\arrow["c"', from=3-2, to=5-2]
	\arrow["f"', from=5-2, to=5-4]
	\arrow["g"', from=5-4, to=5-6]
	\arrow["p"', curve={height=12pt}, from=1-1, to=5-2]
	\arrow["q", curve={height=-6pt}, from=1-1, to=3-6]
	\arrow["{h'}"', dashed, from=1-1, to=3-4]

\begin{proof}
Suppose we have morphisms $C\xleftarrow[]{p}G\xrightarrow[]{q}E$ such that $(g\circ f)\circ p=e\circ q$.

By the pullback square $BEDF$,
\[e\circ q = g\circ (f\circ p)\implies \exists! h':G\rightarrow B, q=b\circ h' \wedge f\circ p = d\circ h'\]
By the pullback square $ABCD$ and using $h'$,
\[d\circ h'=f\circ p \implies \exists! h:G\rightarrow A, p=c\circ h \wedge h'=a\circ h\]
From the unique $h,h'$ such that $q=b\circ h',h'=a\circ h$, 
then $h$ is the unique morphism where $q=b\circ a\circ h$.
Then combining $q=(b\circ a)\circ h$ with $p=c\circ h$ (from pullback square $ABCD$) where $h$ is unique, 
$AECF$ is also a pullback square.
\end{proof}
